CREATE DATABASE IF NOT EXISTS multiorg
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE multiorg;


-- =========================================================
-- 1. ORGANIZATIONS
-- =========================================================
CREATE TABLE IF NOT EXISTS organizations (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    name VARCHAR(255) NOT NULL,
    type VARCHAR(50) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE,

    email VARCHAR(255) NOT NULL,
    phone VARCHAR(50),

    -- Organization Logo
    logo_url VARCHAR(500),

    -- Organization Location
    address VARCHAR(255),
    map_link VARCHAR(500),

    -- Organization Status
    status ENUM('ACTIVE', 'INACTIVE') NOT NULL DEFAULT 'ACTIVE',

    -- Footer Settings
    footer_description TEXT,
    copyright_text VARCHAR(500),

    -- Theme: { "primaryColor": "#4f46e5", "secondaryColor": "#f8fafc" }
    branding JSON,

    -- Hero stats: [{ "value": "27", "label": "INSTITUTIONS", "subLabel": "..." }]
    stats_banner JSON,

    -- Navbar buttons: [{ "name": "ABC College", "link": "https://..." }]
    sister_organizations JSON,

    -- Extra footer block: { "facultyTitle", "facultyDetails", "contactTitle", "contactInfo" }
    footer_config JSON,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
);


-- =========================================================
-- 2. SYSTEM SETTINGS
-- =========================================================
CREATE TABLE IF NOT EXISTS settings (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    system_name VARCHAR(255) NOT NULL,
    system_logo_url VARCHAR(500),
    system_email VARCHAR(255),
    system_phone VARCHAR(50),

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
);


-- =========================================================
-- 3. USERS
-- =========================================================
CREATE TABLE IF NOT EXISTS users (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NULL,

    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,

    -- Profile Photo
    profile_photo_url LONGTEXT,

    -- User Role
    role ENUM('SUPER_ADMIN', 'ORG_ADMIN') NOT NULL,

    -- Increase when password changes/resets
    -- Used to invalidate old JWT sessions
    token_version INT NOT NULL DEFAULT 0,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE
);


-- =========================================================
-- 4. PASSWORD RESET TOKENS
-- =========================================================
CREATE TABLE IF NOT EXISTS password_reset_tokens (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    user_id INT UNSIGNED NOT NULL,

    -- Temporary reset token
    token VARCHAR(255) NOT NULL UNIQUE,

    -- Token expiration time
    expires_at DATETIME NOT NULL,

    -- Prevent token from being reused
    used BOOLEAN NOT NULL DEFAULT FALSE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
);


-- =========================================================
-- 4b. PASSWORD RESET REQUESTS
-- Super admin approval queue fed by POST /api/auth/forgot-password
-- =========================================================
CREATE TABLE IF NOT EXISTS password_reset_requests (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    user_id INT UNSIGNED NULL,
    organization_id INT UNSIGNED NULL,

    -- Denormalized so the queue survives a deleted user/org
    org_name VARCHAR(255),
    admin_name VARCHAR(255),
    admin_email VARCHAR(255) NOT NULL,

    status ENUM('PENDING', 'ACCEPTED', 'REJECTED') NOT NULL DEFAULT 'PENDING',

    -- Reset token, created up-front and only revealed once approved
    token VARCHAR(255),

    handled_by INT UNSIGNED NULL,
    handled_at DATETIME,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE,

    FOREIGN KEY (handled_by)
        REFERENCES users(id)
        ON DELETE SET NULL,

    INDEX idx_reset_requests_status (status)
);


-- =========================================================
-- 5. PAGES
-- Home, About, Contact, Admission, etc.
-- =========================================================
CREATE TABLE IF NOT EXISTS pages (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NOT NULL,

    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL,

    sections JSON,

    -- Navbar dropdown sub-items: ["Teachers", "Gallery"]
    dropdown_items JSON,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE,

    UNIQUE KEY org_page_slug (organization_id, slug)
);


-- =========================================================
-- 6. STAFF
-- =========================================================

    department VARCHAR(255),

    email VARCHAR(255),
    phone VARCHAR(50),

    qualification VARCHAR(500),
    photo_url VARCHAR(500),

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE
);


-- =========================================================
-- 7. NOTICES
-- =========================================================
CREATE TABLE IF NOT EXISTS notices (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NOT NULL,

    title VARCHAR(255) NOT NULL,
    content TEXT,

    image_url VARCHAR(500),

    published BOOLEAN DEFAULT FALSE,
    published_at DATETIME,

    publish_on_main_portal BOOLEAN DEFAULT FALSE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE
);


-- =========================================================
-- 8. EVENTS
-- =========================================================
CREATE TABLE IF NOT EXISTS events (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NOT NULL,

    title VARCHAR(255) NOT NULL,

    image_url VARCHAR(500),

    publish_on_main_portal BOOLEAN DEFAULT FALSE,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

    FOREIGN KEY (organization_id)

-- =========================================================
-- 9. NEWS
-- =========================================================
CREATE TABLE IF NOT EXISTS news (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NOT NULL,

    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL,

    excerpt TEXT,
    content LONGTEXT,

    featured_image VARCHAR(500),

    published BOOLEAN DEFAULT FALSE,
    published_at DATETIME,

    publish_on_main_portal BOOLEAN DEFAULT FALSE,

        REFERENCES organizations(id)
        ON DELETE CASCADE,

-- =========================================================
-- 10. COURSES (DISCIPLINES / PROGRAMS)
-- =========================================================
CREATE TABLE IF NOT EXISTS courses (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NOT NULL,

    name VARCHAR(255) NOT NULL,
    code VARCHAR(100),
    discipline VARCHAR(255),
    description TEXT,
    duration VARCHAR(100),

    status ENUM('ACTIVE', 'INACTIVE') NOT NULL DEFAULT 'ACTIVE',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE
);


-- =========================================================
-- 11. STUDENTS
-- =========================================================
CREATE TABLE IF NOT EXISTS students (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    organization_id INT UNSIGNED NOT NULL,
    course_id INT UNSIGNED NULL,

    name VARCHAR(255) NOT NULL,
    email VARCHAR(255),
    phone VARCHAR(50),
    roll_number VARCHAR(100),

    status ENUM('ACTIVE', 'INACTIVE', 'GRADUATED') NOT NULL DEFAULT 'ACTIVE',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (organization_id)
        REFERENCES organizations(id)
        ON DELETE CASCADE,

    FOREIGN KEY (course_id)
        REFERENCES courses(id)
        ON DELETE SET NULL
);


-- =========================================================
-- 12. CONTACTS
-- =========================================================
CREATE TABLE IF NOT EXISTS contacts (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    phone VARCHAR(50),

    subject VARCHAR(255) NOT NULL,
    message TEXT NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_contacts_created_at (created_at)
);
